### 4. 快速笔记解释器绑定过滤修复

**文件**: `notebook.controller.js`

`filterQuickNoteInterpreterBindings` 原本只保留快速笔记对应的单个解释器，导致段落使用其他解释器时绑定列表不显示。修复后会扫描所有段落的 `%magic` 前缀，保留实际使用到的解释器。

```javascript
// 收集段落实际使用到的解释器名称
let usedNames = new Set();
if ($scope.note && $scope.note.paragraphs) {
  $scope.note.paragraphs.forEach(function(p) {
    let text = p.text || '';
    let magicMatch = text.match(/^\s*%([a-zA-Z0-9_\-]+(?:\.\w+)*)/);
    if (magicMatch) {
      usedNames.add(magicMatch[1].split('.')[0]);
    }
  });
}

// 过滤时保留段落用到的解释器
let filteredBindings = interpreterBindings.filter(function(setting) {
  if (setting && (setting.id === quickNoteBinding || setting.name === quickNoteBinding)) {
    matchedBinding = setting;
    return true;
  }
  if (setting && usedNames.has(setting.name)) {
    return true;
  }
  return false;
});
```

### 5. 重启按钮绑定未加载时自动拉取

**文件**: `paragraph.controller.js`

`interpreterBindings` 是懒加载的（只在打开设置面板时拉取），导致首次点击重启按钮时可能找不到绑定。修复后会自动触发加载并等待结果。

```javascript
function ensureInterpreterBindingsLoaded() {
  let scope = $scope;
  while (scope && !scope.interpreterBindings) {
    scope = scope.$parent;
  }
  if (scope && scope.interpreterBindings && scope.interpreterBindings.length > 0) {
    return $q.when();
  }
  let noteId = getCurrentNoteId();
  websocketMsgSrv.getInterpreterBindings(noteId);
  return $q(function(resolve) {
    let deregister = $rootScope.$on('interpreterBindings', function() {
      deregister();
      resolve();
    });
  });
}
```
